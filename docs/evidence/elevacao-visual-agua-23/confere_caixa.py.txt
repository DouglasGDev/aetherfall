#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""Re-deriva a regua da sonda de pixel (dossie 23 §5.2) a partir dos recortes
arquivados, nao dos frames vivos.  E o motivo de os PNGs guardados serem a caixa de
192 px em torno do centro da agua, com o passo 2: as 9 216 amostras que o probe somou
sao exatamente os bytes deste recorte, entao quem nao tem a arvore, nem Godot, nem GPU
confer o numero publicado.

A aritmetica e a do probe, e a precisao tambem — e aqui esta o detalhe que faz desta uma
testemunha em vez de uma reimpressao.  O `.gd` compara `absf(ca.r - cb.r)` com
`1.0 / 255.0` sobre canais que o `Image` devolve em **float32**.  O float32 mais
proximo de 1/255 vale 0,0039215689 e a subtracao de dois bytes vizinhos (8/255 - 7/255)
vale 0,0039215686: a amostra cuja diferenca e exatamente um byte **nao** passa no >=.
Medido aqui, na caixa da agua desta perna: em byte puro (|delta| >= 1) sairiam
ge1=9141 ge3=8402; com a aritmetica do probe saem ge1=8867 ge3=7807 — os dois numeros
que a corrida imprimiu.  Quem reimplementar a regua em inteiros, portanto, obtem uma
contagem maior que a publicada, e o conservadorismo e do instrumento, nao da cena: o
`fracao_ge3` publicado e um piso.
"""
import sys
import numpy as np
from PIL import Image

PASSO = 2


def load(path):
    """A caixa ja recortada, em float32 sobre 255 — o que `get_pixel` devolve."""
    a = np.asarray(Image.open(path).convert("RGB"), dtype=np.uint8)
    return a[::PASSO, ::PASSO, :].astype(np.float32) / np.float32(255.0)


def diff(a_path, b_path):
    d = np.abs(load(a_path) - load(b_path)).astype(np.float32)
    mx = d.max(axis=2).astype(np.float32)
    um = np.float32(1.0) / np.float32(255.0)
    tres = np.float32(3.0) / np.float32(255.0)
    n = int(d.shape[0] * d.shape[1])
    return {"n": n,
            "ge1": int((mx >= um).sum()),
            "ge3": int((mx >= tres).sum()),
            "mr": float(d[:, :, 0].mean()),
            "mg": float(d[:, :, 1].mean()),
            "mb": float(d[:, :, 2].mean()),
            "size": (d.shape[1], d.shape[0])}


def main():
    if len(sys.argv) != 3:
        print("uso: confere_caixa.py <caixa-t1.png> <caixa-t2.png>", file=sys.stderr)
        return 2
    r = diff(sys.argv[1], sys.argv[2])
    print("CAIXA  %s vs %s  tamanho=%dx%d" % (
        sys.argv[1], sys.argv[2], r["size"][0], r["size"][1]))
    print("CAIXA  amostras=%d  ge1=%d  ge3=%d  fracao_ge3=%.4f  "
          "media_canal r=%.6f g=%.6f b=%.6f" % (
              r["n"], r["ge1"], r["ge3"], r["ge3"] / float(max(r["n"], 1)),
              r["mr"], r["mg"], r["mb"]))
    return 0


if __name__ == "__main__":
    sys.exit(main())
